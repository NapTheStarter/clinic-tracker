// วางโค้ดนี้ทับใน Code.gs ทั้งหมด // จากนั้น Deploy > New deployment > Web app // Execute as: Me // Who has access: Anyone const SHEET_NAME = "entries"; const HEADERS = ["timestamp", "cn", "channelCode", "channelLabel", "name", "date", "note"]; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } const firstCell = sheet.getRange(1, 1).getValue(); if (firstCell !== HEADERS[0]) { // หัวตารางหาย/ไม่ตรง หรือชีทว่างเปล่า -> แทรกแถวหัวตารางไว้บนสุดเสมอ sheet.insertRowBefore(1); sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); sheet.setFrozenRows(1); } return sheet; } function doPost(e) { const sheet = getSheet_(); const p = (e && e.parameter) || {}; if (p.action === "delete") { return deleteRow_(sheet, p.id); } const row = [ new Date(), String(p.cn || ""), p.channelCode || "", p.channelLabel || "", p.name || "", p.date || "", p.note || "", ]; sheet.appendRow(row); return jsonOut_({ ok: true }); } function doGet(e) { const sheet = getSheet_(); const values = sheet.getDataRange().getValues(); const headers = values[0]; const rows = values.slice(1).map(function (r, idx) { const obj = { rowIndex: idx + 2 }; headers.forEach(function (h, i) { obj[h] = r[i] instanceof Date ? r[i].toISOString() : r[i]; }); return obj; }); return jsonOut_({ ok: true, entries: rows }); } function deleteRow_(sheet, rowIndex) { if (!rowIndex || rowIndex < 2) return jsonOut_({ ok: false, error: "bad row" }); sheet.deleteRow(Number(rowIndex)); return jsonOut_({ ok: true }); } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType( ContentService.MimeType.JSON ); }
